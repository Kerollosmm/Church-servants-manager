## Phase 2: Implementation Plan

### Fix Group A — Security (C1, C2) 🔴 CRITICAL

#### A1. Firestore Security Rules — `firestore.rules` (new)

```
// firestore.rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() { return request.auth != null; }
    function userRole() {
      return get(/databases/$(database)/documents/Users/$(request.auth.uid)).data.get('role','');
    }
    function isAdmin() { return isSignedIn() && userRole() == 'admin'; }
    function isServant() { return isSignedIn() && userRole() == 'servant'; }
    function isServantOfTeam(teamId) {
      return isServant() &&
        teamId in get(/databases/$(database)/documents/Users/$(request.auth.uid)).data.get('assignedTeamIds',[]);
    }

    // Users collection
    match /Users/{uid} {
      allow read: if isSignedIn() && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }

    // Students collection
    match /Students/{docId} {
      allow read: if isSignedIn();
      allow create, update, delete: if isAdmin();
    }

    // Classes + attendance sessions + marks
    match /Classes/{teamId} {
      allow read: if isSignedIn();
      allow create, update, delete: if isAdmin();

      match /attendance_sessions/{sessionId} {
        allow read: if isSignedIn();
        allow create, update, delete: if isAdmin();

        match /marks/{studentId} {
          allow read: if isSignedIn();
          allow create, update, delete: if isAdmin() || isServantOfTeam(teamId);
        }
      }
    }
  }
}
```

**Testing**: Deploy to Firebase Emulator; run rule unit tests for each role/path.
